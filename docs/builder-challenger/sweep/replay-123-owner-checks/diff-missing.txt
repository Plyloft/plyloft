+import { describe, expect, it } from "vitest";
+import { densityClass, readDensity } from "./boardDensity";
+
+describe("board density", () => {
+  it("reads compact and falls back to comfortable", () => {
+    expect(readDensity("compact")).toBe("compact");
+    expect(readDensity("comfortable")).toBe("comfortable");
+    expect(readDensity(null)).toBe("comfortable");
+    expect(readDensity("garbage")).toBe("comfortable");
+  });
+  it("maps a density to the card list's class", () => {
+    expect(densityClass("compact")).toBe("board-cards board-cards--compact");
+    expect(densityClass("comfortable")).toBe("board-cards");
+  });
+});
